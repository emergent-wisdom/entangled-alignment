% Figure: Entangled Alignment training path and evidence boundary
% Layout: one Teacher-side spine (four uniform stages), the Reader Core as the
% single conditioning satellite, graph recurrence as a feedback arc, and two
% parallel status banners bracketing the evidence boundary.
\begin{tikzpicture}[
    box/.style={draw=black!70, rounded corners=3pt, align=center, font=\small, line width=0.8pt},
    stagebox/.style={box, minimum height=2.45cm, text width=3.1cm},
    teacher/.style={stagebox, fill=blue!6},
    artifact/.style={stagebox, fill=green!7},
    student/.style={box, fill=purple!10, text width=3.6cm, minimum height=1.7cm},
    corenode/.style={box, fill=purple!8, draw=purple!55, dashed, text width=3.2cm, minimum height=1.1cm},
    banner/.style={box, text width=11.8cm, minimum height=0.85cm, font=\footnotesize},
    arrow/.style={->, >=Stealth, thick, black!70},
    condition/.style={->, >=Stealth, dashed, thick, purple!60}
    ]

    % Teacher-side spine
    \node[teacher]  (source)    at (-6.3, 0) {Ordered\\source span\\$x_t$\\\footnotesize + source provenance};
    \node[teacher]  (update)    at (-2.1, 0) {Synthetic\\reader update\\$u_t$\\\footnotesize questions, uncertainty,\\revision};
    \node[artifact] (graph)     at (2.1, 0)  {Understanding\\Graph\\$g_t$\\\footnotesize typed state +\\supersession};
    \node[artifact] (serialize) at (6.3, 0)  {Candidate\\serialization\\$z$\\\footnotesize source + update\\+ graph layer};
    \draw[arrow] (source) -- (update);
    \draw[arrow] (update) -- (graph);
    \draw[arrow] (graph) -- (serialize);

    % Conditioning: recurrent Core above; prior graph state ($g_{t-1}$) fed
    % back to the update as a compact recurrence arc between the two boxes
    \node[corenode] (core) at (-2.1, 2.35) {Reader Core $c$\\\footnotesize recurrent\\evaluative reference};
    \draw[condition] (core) -- (update);
    \draw[condition] (graph.160) to[out=150,in=30] node[midway, above=1pt, font=\scriptsize, purple!60] {$g_{t-1}$} (update.20);

    % Status banner above the boundary
    \node[banner, fill=green!7, draw=green!45!black] (demonstrated) at (0, -2.05) {\textbf{Current evidence:} Teacher-side graph, synthesis, and prose artifacts are implemented and \mbox{inspectable}.};
    \draw[dashed, gray!70] (-8.1, -2.85) -- (8.1, -2.85);
    \node[font=\scriptsize\bfseries, text=black!70, fill=white, inner sep=2pt] at (0, -2.85) {evidence boundary};

    % Proposed Student path below the boundary
    \node[student] (student) at (3.4, -4.25) {Candidate Student\\$p_\theta$\\\footnotesize masked/weighted training};
    \node[student] (tests)   at (-3.4, -4.25) {Ablations and evaluations\\\footnotesize H1 trace value\\H2 graph advantage\\H3 Core stability};
    \draw[arrow, rounded corners=8pt] (serialize.south) |- (student.east);
    \draw[arrow] (student) -- (tests);

    % Status banner below the boundary, parallel to the one above
    \node[banner, fill=purple!7, draw=purple!45] (untested) at (0, -6.0) {\textbf{Untested effects:} Student learning, stable \mbox{orientation}, safety improvement, and removal resistance.};

\end{tikzpicture}
