% Figure: The Multi-Agent Metabolic Cycle (one annotation step)
% Layout: horizontal spine from Reader Core to Translator through the shared
% graph; Workers orbit the graph in a symmetric four-position ring, each with a
% dashed read arrow out and a solid write arrow back. The ghost node marks the
% named Workers as an illustrative subset.
\begin{tikzpicture}[
    block/.style={rectangle, draw=black!70, fill=white, text width=6em, text centered, rounded corners=3pt, minimum height=3em, font=\small, line width=0.8pt},
    ghost/.style={block, dashed, draw=black!40, fill=gray!4, font=\small\itshape, text=black!55},
    hub/.style={ellipse, draw=black!70, fill=gray!10, minimum width=3.0cm, minimum height=1.8cm, align=center, font=\small\bfseries, line width=0.8pt},
    core/.style={rectangle, draw=purple!55, dashed, fill=purple!8, text width=6.5em, text centered, rounded corners=3pt, minimum height=3em, font=\small},
    arrow/.style={thick, ->, >=Stealth, black!70},
    readarrow/.style={arrow, dashed, black!45},
    corearrow/.style={arrow, dashed, draw=purple!60}
]
    % Spine: orientation -> annotation -> selection -> expression
    \node[hub]   (graph)  at (0, 0)    {Shared\\Graph\\State};
    \node[core]  (core)   at (-7.7, 0) {Reader Core\\$c^{(v)}$\\\scriptsize recurrent reference};
    \node[block] (reader) at (-4.3, 0) {\textbf{Reader}\\(Trigger)};
    \node[block] (synth)  at (4.3, 0)  {\textbf{Synthesizer}\\(The Pull)};
    \node[block] (trans)  at (7.7, 0)  {\textbf{Translator}\\(Expansion)};

    % Conditioning is distributed in the reference implementation; placement
    % remains an ablation rather than a necessity claim.
    \draw[corearrow] (core) -- (reader);
    \draw[arrow] (reader) -- node[above, font=\scriptsize] {Content} (graph);
    \draw[arrow] (graph) -- node[above, font=\scriptsize] {Read} (synth);
    \draw[arrow] (synth) -- (trans);

    % Worker ring around the graph (illustrative subset of the full roster)
    \node[block] (worker1) at (-2.4, 2.3)  {Axiologist};
    \node[block] (worker2) at (2.4, 2.3)   {Skeptic};
    \node[block] (worker3) at (-2.4, -2.3) {Psychologist};
    \node[ghost] (worker4) at (2.4, -2.3)  {$\cdots$ further\\Worker roles};

    % Read (dashed, outward) / write (solid, inward) pairs, mirrored per quadrant
    % Write labels sit on the outer side of the solid arrows, clear of the
    % dashed read arrows, mirrored about the figure's vertical axis
    \draw[readarrow] (graph) to[bend left=14] (worker1);
    \draw[arrow] (worker1) to[bend left=14] node[midway, sloped, above=1.5pt, font=\scriptsize] {Mutate} (graph);
    \draw[readarrow] (graph) to[bend right=14] (worker2);
    \draw[arrow] (worker2) to[bend right=14] node[midway, sloped, above=1.5pt, font=\scriptsize] {Critique} (graph);
    \draw[readarrow] (graph) to[bend right=14] (worker3);
    \draw[arrow] (worker3) to[bend right=14] (graph);
    \draw[readarrow] (graph) to[bend left=14] (worker4);
    \draw[arrow, black!45] (worker4) to[bend left=14] (graph);

\end{tikzpicture}
